% columns: widths as fractions of \textwidth; [T] aligns their tops.
% \showdividers in main.tex outlines every column and row with its size.
\begin{frame}{Columns}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \begin{block}{Column 1}
        \cmd{begin\{column\}\{0.32\textbackslash textwidth\}}
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Column 2}
        Widths are shares of the text width.
      \end{block}
    \end{column}
    \begin{column}{0.32\textwidth}
      \begin{block}{Column 3}
        \texttt{[T]} aligns the tops.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
