% Starter deck and tutorial for titletemplates.sty + tex2html.
%   PDF:   latexmk -pdf main.tex          (biblatex + biber)
%   HTML:  tex2html                        (-> slides.html; --watch to rebuild on save)
% Every section in sections/ shows one element with a working example; the
% comments above each frame explain its options. Copy the frames you need
% into your own sections, replace the [placeholders] below, and delete the
% tutorial sections.
\documentclass[aspectratio=169,11pt]{beamer}   % 8pt ... 20pt scales all text

\usepackage[T1]{fontenc}
\usepackage{newtxtext,newtxmath}   % Times-like serif
\usefonttheme{serif}
\setbeamertemplate{navigation symbols}{}
\usepackage{titletemplates}        % installed/updated here by tex2html
\usepackage{animate}               % animations from image sequences

% ---------------------------------------------------------------- references
% numeric-comp: [1], [2-4]; style=authoryear for (Author, Year)
\usepackage[backend=biber, style=numeric-comp, sorting=none,
            maxbibnames=3, giveninits=true, doi=false, url=false]{biblatex}
\addbibresource{references.bib}

% ---------------------------------------------------------------- title slide fields
\SetTitle{[Presentation Title]}
\SetSeminar{[Seminar Name]}            % e.g. Extension seminar
\SetDegree{[Degree Name]}              % e.g. Doctor of Philosophy
\SetPresenter{[Presenter Name]}
\SetRollNo{[Roll Number]}
\SetSupervisor{[Supervisor Name]}
\SetDepartment{[Department Name]}
\SetInstitute{[Institute Name]}
\SetLogo{figures/logo}                 % path without extension; a placeholder box until it exists

\SetAuthorOne{[Author 1]}              % conference title slide
\SetAuthorTwo{[Author 2]}
\SetConference{[Conference Name]}

\SetShortTitle{[Short Title]}          % footer of content slides
\SetFootAuthor{[Presenter Name]}       % footer of content slides

% ---------------------------------------------------------------- content slide options
% Each switch also works between frames, for the frames after it.
\TitleBarOn                % \TitleBarOff: no blue bar, underlined black title
\SetTitleHeight{1.1cm}     % height of the title area / blue bar
\SectionNameOn             % \SectionNameOff: no section name at the top right
\FooterOn                  % \FooterNumberOnly: slide number only; \FooterOff: no footer
% \SetTextMargins{1cm}{1cm} % slide body side margins (also between frames)
% \SetBarMargin{0.4cm}      % side margin of the title bar and footer
% \SetCaptionSize{\footnotesize} % caption size for the whole deck (default \small)
% \showdividers            % editing aid: outline rows and columns and show the
%                          % annotate grid (\hidedividers)

\begin{document}

\SeminarTitleFrame         % thesis / seminar title slide
% \ConferenceTitleFrame    % conference title slide (two authors)

\OverviewFrame             % lists the \section names below

\input{sections/01-text}             % lists, text styles, math, blocks
\input{sections/02-layout}           % columns, rows (grids), margins
\input{sections/03-figures}          % figure + caption + label, placeholders
\input{sections/04-tables}           % slidetable styles
\input{sections/05-animations}       % animation: one row, grid (columns=)
\input{sections/06-shapes}           % annotate + \shape, \slideshape
\input{sections/07-overlays}         % \pause, \item<2->, \only, \uncover, \alt
\input{sections/08-citations}        % \cite, \textcite, litreview
\input{sections/09-research}         % progress timeline, publications
\input{sections/10-slide-options}    % title bar / footer switches, speaker notes

\ReferencesFrame           % the cited entries; long lists continue on more slides

\ThankYouFrame

\end{document}
